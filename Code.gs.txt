// アプリ（GitHub Pages）からの窓口
function doPost(e) {
  let req = {};
  try { req = JSON.parse(e.postData.contents); } catch (err) {}
  if (req && Array.isArray(req.events)) { try { lineHook_(req); } catch (err) {} return ContentService.createTextOutput('OK'); }
  return ContentService.createTextOutput(JSON.stringify(api(req)))
    .setMimeType(ContentService.MimeType.JSON);
}

function api(req) {
  try {
    req = req || {};
    if (/^g[A-Z]/.test(String(req.action))) return guestApi_(req);
    const pass = props_().getProperty('APP_PASS');
    if (!pass) return { ok: false, error: 'スクリプトプロパティ APP_PASS が未設定です' };
    if (req.pass !== pass) return { ok: false, auth: true, error: 'パスコードが違います' };
    switch (req.action) {
      case 'ping': return { ok: true };
      case 'translate': return Object.assign({ ok: true }, translate_(req));
      case 'back': return Object.assign({ ok: true }, back_(req));
      case 'saveEx': return Object.assign({ ok: true }, saveExample_(req));
      case 'deleteEx': return Object.assign({ ok: true }, deleteExample_(req));
      case 'useEx': return Object.assign({ ok: true }, useExample_(req));
      case 'list': return Object.assign({ ok: true }, list_());
      case 'saveTerm': return Object.assign({ ok: true }, saveTerm_(req));
      case 'deleteTerm': return Object.assign({ ok: true }, deleteTerm_(req));
      case 'sync': return Object.assign({ ok: true }, syncGlossary_());
      case 'usage': return Object.assign({ ok: true }, usage_());
      case 'ocr': return Object.assign({ ok: true }, ocr_(req));
      case 'history': return Object.assign({ ok: true }, history_());
      case 'setAuto': props_().setProperty('AUTO_REG', req.on ? '1' : '0'); return { ok: true, auto: !!req.on };
      case 'promote': return Object.assign({ ok: true }, promote_(req));
      case 'dismiss': return Object.assign({ ok: true }, dismiss_(req));
      case 'rules': return Object.assign({ ok: true }, listRules_());
      case 'saveRule': return Object.assign({ ok: true }, saveRule_(req));
      case 'deleteRule': return Object.assign({ ok: true }, deleteRule_(req));
      case 'moveRule': return Object.assign({ ok: true }, moveRule_(req));
      case 'setStore': props_().setProperty('STORE_NAME', String(req.name || '').slice(0, 60)); return { ok: true };
      case 'uploadPhoto': return Object.assign({ ok: true }, uploadPhoto_(req));
      case 'photo': return Object.assign({ ok: true }, photo_(req));
      case 'archiveLog': return Object.assign({ ok: true }, archiveLog_());
      case 'stays': return Object.assign({ ok: true }, listStays_(req));
      case 'createStay': return Object.assign({ ok: true }, createStay_(req));
      case 'closeStay': return Object.assign({ ok: true }, closeStay_(req));
      case 'chat': return Object.assign({ ok: true }, staffChat_(req));
      case 'staffSend': return Object.assign({ ok: true }, staffSend_(req));
      case 'getInfo': return Object.assign({ ok: true }, getInfo_());
      case 'setInfo': return Object.assign({ ok: true }, setInfo_(req));
      case 'lineStatus': return Object.assign({ ok: true }, lineStatus_());
      case 'lineSetTarget': return Object.assign({ ok: true }, lineSetTarget_(req));
      case 'lineTest': return Object.assign({ ok: true }, lineTest_());
      case 'stats': return Object.assign({ ok: true }, stats_());
      case 'templates': return Object.assign({ ok: true }, listTpl_());
      case 'saveTpl': return Object.assign({ ok: true }, saveTpl_(req));
      case 'deleteTpl': return Object.assign({ ok: true }, deleteTpl_(req));
      case 'phrases': return Object.assign({ ok: true }, listPhrases_());
      case 'savePhrase': return Object.assign({ ok: true }, savePhrase_(req));
      case 'deletePhrase': return Object.assign({ ok: true }, deletePhrase_(req));
      case 'movePhrase': return Object.assign({ ok: true }, movePhrase_(req));
      default: return { ok: false, error: '不明な操作です' };
    }
  } catch (e) {
    return { ok: false, error: String((e && e.message) || e) };
  }
}

/* ---------- 共通 ---------- */
function props_() { return PropertiesService.getScriptProperties(); }
function book_() {
  const b = SpreadsheetApp.getActiveSpreadsheet();
  if (b) return b;
  const id = props_().getProperty('SHEET_ID');
  if (!id) throw new Error('保存先のスプレッドシートが見つかりません（SHEET_ID を設定してください）');
  return SpreadsheetApp.openById(id);
}
function sheet_(name, head) {
  const b = book_();
  let sh = b.getSheetByName(name);
  if (!sh) {
    sh = b.insertSheet(name);
    sh.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold');
    sh.setFrozenRows(1);
    sh.getRange(1, 1, sh.getMaxRows(), Math.min(head.length, 5)).setNumberFormat('@');
  }
  return sh;
}
function norm_(s) {
  let t = String(s || '').normalize('NFKC').toLowerCase().replace(/\s+/g, ' ');
  // 日本語・中国語・韓国語の文は、間の空白の有無を区別しない
  if (/[぀-ヿ㐀-鿿가-힯]/.test(t)) t = t.replace(/ /g, '');
  return t.replace(/[。．.!！?？、,\s]+$/, '').trim();
}
function safe_(s) { s = String(s || ''); return /^[=+\-@]/.test(s) ? "'" + s : s; }
function clean_(s) { return String(s || '').replace(/[\t\r\n]+/g, ' ').trim(); }
function grams_(s) {
  const n = norm_(s).replace(/\s/g, ''), m = {};
  for (let i = 0; i < n.length - 1; i++) { const g = n.substr(i, 2); m[g] = (m[g] || 0) + 1; }
  return m;
}
function dice_(a, b) {
  let inter = 0, ca = 0, cb = 0;
  for (const k in a) { ca += a[k]; if (b[k]) inter += Math.min(a[k], b[k]); }
  for (const k in b) cb += b[k];
  return ca + cb ? 2 * inter / (ca + cb) : 0;
}
function withLock_(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try { return fn(); } finally { lock.releaseLock(); }
}

/* ---------- DeepL ---------- */
function deepl_(method, path, body) {
  const key = String(props_().getProperty('DEEPL_KEY') || '').trim();
  if (!key) throw new Error('スクリプトプロパティ DEEPL_KEY が未設定です');
  const host = /:fx$/.test(key) ? 'https://api-free.deepl.com' : 'https://api.deepl.com';
  const opt = { method: method, headers: { Authorization: 'DeepL-Auth-Key ' + key }, muteHttpExceptions: true };
  if (body) { opt.contentType = 'application/json'; opt.payload = JSON.stringify(body); }
  // 混み合い（429）や一時的なエラー（5xx）は、少し待って最大3回やり直す
  let res, code, txt;
  for (let i = 0; i < 4; i++) {
    res = UrlFetchApp.fetch(host + path, opt);
    code = res.getResponseCode(); txt = res.getContentText();
    if (code !== 429 && code < 500) break;
    if (i < 3) Utilities.sleep(1500 * Math.pow(2, i));
  }
  if (code === 456) throw new Error(/glossar/.test(path) ? 'DeepLの用語集の上限に達しました（無料版は用語集1つまで）' : 'DeepLの今月の無料枠を使い切りました');
  if (code === 403) throw new Error('DeepLのAPIキーが正しくありません');
  if (code === 429) throw new Error('DeepLが混み合っています。少し待ってからもう一度');
  if (code >= 300) throw new Error('DeepLエラー ' + code + ': ' + txt.slice(0, 200));
  return txt ? JSON.parse(txt) : {};
}

function translate_(q) {
  const text = String(q.text || '').trim();
  if (!text) throw new Error('文章が空です');
  const dir = dir_(q.dir, text, q.lang);
  const L = dir.split('-');
  const ex = examples_();
  const nt = norm_(text);

  // ゲストの言葉を訳すとき（会話）は、店のルールから答えの候補を探す
  const hints = function (ja) { return q.log && L[1] === 'ja' ? ruleHints_(L[0], text, ja) : []; };

  // 1) 覚えた用例と同じ文なら、その訳をそのまま使う（DeepLの文字数を使わない）
  if (!q.skipMemory) {
    const hit = ex.find(function (r) { return r.dir === dir && norm_(r.src) === nt; });
    if (hit) { bump_(hit); stat_('mem'); if (q.log || q.alog) alog_(dir, text, hit.tgt, 'memory', q.log ? '会話' : '文章'); return { text: hit.tgt, from: 'memory', dir: dir, id: hit.id, similar: [], rules: hints(hit.tgt) }; }
    // 1.5) よく似た文を覚えていれば、DeepLに送る前にスタッフに確かめる
    if (!q.noNear) {
      const near = nearest_(ex, dir, text);
      if (near) return { text: '', from: 'near', dir: dir, near: near, rules: hints(L[1] === 'ja' ? near.tgt : '') };
    }
  }

  // 2) DeepL（用語集と直前の会話を添える）
  const body = { text: [text], source_lang: LANG[L[0]].src, target_lang: LANG[L[1]].tgt };
  const gid = glossId_(dir);
  if (gid) body.glossary_id = gid;
  if (q.context) body.context = String(q.context).slice(0, 3000);
  const r = deeplT_(body);
  stat_('deepl');

  // 3) 似ている用例を参考として返す
  const g = grams_(text);
  const similar = ex.filter(function (x) { return x.dir === dir && norm_(x.src) !== nt; })
    .map(function (x) { return { x: x, s: dice_(g, grams_(x.src)) }; })
    .filter(function (o) { return o.s >= 0.45; })
    .sort(function (a, b) { return b.s - a.s; }).slice(0, 3)
    .map(function (o) { return { id: o.x.id, src: o.x.src, tgt: o.x.tgt, score: Math.round(o.s * 100) }; });

  const out = r.translations[0].text;
  const auto = q.log ? logTurn_(dir, text, out) : false;
  if (q.log || q.alog) alog_(dir, text, out, 'deepl', q.log ? '会話' : '文章');
  return { text: out, from: 'deepl', dir: dir, similar: similar, auto: auto, rules: hints(out) };
}

// 用語集を付けて訳す。用語集が使えない言語の組み合わせなら、用語集なしで訳し直す
function deeplT_(body) {
  try { return deepl_('post', '/v2/translate', body); }
  catch (e) {
    if (body.glossary_id && /DeepLエラー 4\d\d/.test(String(e.message))) {
      const b = JSON.parse(JSON.stringify(body)); delete b.glossary_id;
      return deepl_('post', '/v2/translate', b);
    }
    throw e;
  }
}

// よく似た登録済みの文（数字が違うもの・短すぎるものは対象外）
const NEAR_MIN = 0.8;
function digits_(s) { return (String(s).normalize('NFKC').match(/\d+/g) || []).join(','); }
function nearest_(ex, dir, text) {
  const nt = norm_(text);
  if (nt.replace(/\s/g, '').length < 6) return null;
  const g = grams_(text), dg = digits_(text);
  let best = null;
  ex.forEach(function (x) {
    if (x.dir !== dir || norm_(x.src) === nt || digits_(x.src) !== dg) return;
    const ratio = norm_(x.src).length / nt.length;
    if (ratio < 0.7 || ratio > 1.4) return;
    const s = dice_(g, grams_(x.src));
    if (s >= NEAR_MIN && (!best || s > best.s)) best = { x: x, s: s };
  });
  return best ? { id: best.x.id, src: best.x.src, tgt: best.x.tgt, score: Math.round(best.s * 100) } : null;
}

// ゲストの質問に近い店舗ルール（最大2件）
function ruleHints_(lang, src, ja) {
  try {
    const list = rules_().filter(guestVisible_);
    if (!list.length) return [];
    const gs = grams_(src), gj = ja ? grams_(ja) : null;
    return list.map(function (r) {
      const s = Math.max(gj ? dice_(gj, grams_(r.ja)) : 0, r[lang] ? dice_(gs, grams_(r[lang])) : 0);
      return { r: r, s: s };
    }).filter(function (o) { return o.s >= 0.35; })
      .sort(function (a, b) { return b.s - a.s; }).slice(0, 2)
      .map(function (o) { return { id: o.r.id, ja: o.r.ja, tr: o.r[lang] || o.r.en, say: o.r.say[lang] || '', photos: o.r.photos }; });
  } catch (e) { return []; }
}

/* ---------- 学習の記録（月ごと：覚えた訳で答えた回数／新しく訳した回数） ---------- */
function month_() { return Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM'); }
function stat_(kind) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(3000)) return;
  try {
    const P = props_();
    let st = {};
    try { st = JSON.parse(P.getProperty('STATS') || '{}'); } catch (e) {}
    const m = month_();
    st[m] = st[m] || { mem: 0, deepl: 0 };
    st[m][kind] = (st[m][kind] || 0) + 1;
    const keys = Object.keys(st).sort();
    while (keys.length > 24) delete st[keys.shift()];
    P.setProperty('STATS', JSON.stringify(st));
  } catch (e) {} finally { lock.releaseLock(); }
}
function stats_() {
  let st = {};
  try { st = JSON.parse(props_().getProperty('STATS') || '{}'); } catch (e) {}
  const months = Object.keys(st).sort().reverse().slice(0, 6).map(function (k) {
    return { month: k, mem: st[k].mem || 0, deepl: st[k].deepl || 0 };
  });
  const ex = examples_();
  return {
    month: month_(), months: months,
    examples: ex.length, auto: ex.filter(function (x) { return x.cat === '自動登録'; }).length,
    terms: terms_().length, rules: rules_().length, templates: tpls_().length
  };
}

// 対応言語（DeepLの言語コード）
const LANG = {
  ja: { src: 'JA', tgt: 'JA' },
  en: { src: 'EN', tgt: 'EN-US' },
  zh: { src: 'ZH', tgt: 'ZH-HANS' },
  tw: { src: 'ZH', tgt: 'ZH-HANT' },
  ko: { src: 'KO', tgt: 'KO' },
  // 会話タブで国旗から選べる追加の言語（ルール・フレーズ・ゲスト用ページは上の4言語のまま）
  fr: { src: 'FR', tgt: 'FR' },
  de: { src: 'DE', tgt: 'DE' },
  it: { src: 'IT', tgt: 'IT' },
  es: { src: 'ES', tgt: 'ES' },
  ru: { src: 'RU', tgt: 'RU' },
  th: { src: 'TH', tgt: 'TH' },
  vi: { src: 'VI', tgt: 'VI' },
  id: { src: 'ID', tgt: 'ID' }
};
function dir_(d, text, lang) {
  const m = /^([a-z]{2})-([a-z]{2})$/.exec(String(d || ''));
  if (m && LANG[m[1]] && LANG[m[2]] && (m[1] === 'ja') !== (m[2] === 'ja')) return d;
  const g = LANG[lang] && lang !== 'ja' ? lang : 'en';
  if (/[぀-ヿ]/.test(text)) return 'ja-' + g;          // かな → 日本語
  if (/[가-힯]/.test(text)) return 'ko-ja';            // ハングル
  if (/[一-鿿]/.test(text)) return (g === 'tw' ? 'tw' : 'zh') + '-ja'; // 漢字のみ → 中国語
  return 'en-ja';
}

// スタッフの確認用：訳文を日本語に戻す（覚えた訳・用語集は使わない）
function back_(q) {
  const text = String(q.text || '').trim();
  if (!text) throw new Error('文章が空です');
  const lang = LANG[q.lang] && q.lang !== 'ja' ? q.lang : 'en';
  const r = deepl_('post', '/v2/translate', { text: [text], source_lang: LANG[lang].src, target_lang: 'JA' });
  return { text: r.translations[0].text };
}

function usage_() {
  const r = deepl_('get', '/v2/usage');
  return { count: r.character_count, limit: r.character_limit };
}

/* ---------- 用例（覚えた訳） ---------- */
function examples_() {
  const v = sheet_(SH_EX, HEAD_EX).getDataRange().getValues();
  return v.slice(1).map(function (r, i) {
    return {
      row: i + 2, id: String(r[0]), dir: String(r[1]), src: String(r[2]), tgt: String(r[3]),
      cat: String(r[4] || 'その他'), count: Number(r[5]) || 0,
      updated: r[6] instanceof Date ? r[6].getTime() : 0
    };
  }).filter(function (r) { return r.id && r.src; });
}
function bump_(hit) {
  sheet_(SH_EX, HEAD_EX).getRange(hit.row, 6).setValue(hit.count + 1);
}
function useExample_(q) {
  const hit = examples_().find(function (r) { return r.id === q.id; });
  if (hit) { bump_(hit); stat_('mem'); if (q.log) alog_(hit.dir, hit.src, hit.tgt, 'memory', String(q.log) === 'near' ? '会話' : 'よく使う文'); }
  return {};
}
function saveExample_(q) {
  const src = String(q.src || '').trim(), tgt = String(q.tgt || '').trim();
  if (!src || !tgt) throw new Error('原文と訳文の両方が必要です');
  const dir = /^(ja-(en|zh|tw|ko)|(en|zh|tw|ko)-ja)$/.test(String(q.dir)) ? q.dir : 'ja-en';
  return withLock_(function () {
    const sh = sheet_(SH_EX, HEAD_EX), ex = examples_();
    let hit = q.id ? ex.find(function (r) { return r.id === q.id; }) : null;
    if (!hit) hit = ex.find(function (r) { return r.dir === dir && norm_(r.src) === norm_(src); });
    const now = new Date();
    if (hit) {
      const cat = q.cat || hit.cat;
      sh.getRange(hit.row, 1, 1, 7).setValues([[hit.id, dir, safe_(src), safe_(tgt), cat, hit.count, now]]);
      return { id: hit.id, updated: true };
    }
    const id = Utilities.getUuid().slice(0, 8);
    sh.appendRow([id, dir, safe_(src), safe_(tgt), q.cat || 'その他', 0, now]);
    return { id: id, updated: false };
  });
}
function deleteExample_(q) {
  return withLock_(function () {
    const hit = examples_().find(function (r) { return r.id === q.id; });
    if (hit) sheet_(SH_EX, HEAD_EX).deleteRow(hit.row);
    return {};
  });
}

/* ---------- 用語集（DeepLに教える単語） ---------- */
function termSheet_() {
  const sh = sheet_(SH_TERM, HEAD_TERM);
  // 以前の3列の表に、中国語・韓国語の列を足す
  const head = sh.getDataRange().getValues()[0] || [];
  if (head.length < HEAD_TERM.length || !head[3]) sh.getRange(1, 1, 1, HEAD_TERM.length).setValues([HEAD_TERM]).setFontWeight('bold');
  return sh;
}
function terms_() {
  const v = termSheet_().getDataRange().getValues();
  return v.slice(1).map(function (r, i) {
    return {
      row: i + 2, ja: String(r[0] || '').trim(), en: String(r[1] || '').trim(), memo: String(r[2] || ''),
      zh: String(r[3] || '').trim(), tw: String(r[4] || '').trim(), ko: String(r[5] || '').trim()
    };
  }).filter(function (t) { return t.ja; });
}
// 日本語は必須。訳が空の言語は、その言葉をDeepLで訳して埋める（あとで直せる）
function saveTerm_(q) {
  const ja = clean_(q.ja);
  if (!ja) throw new Error('日本語の言葉を入れてください');
  const cur = terms_().find(function (x) { return x.ja === clean_(q.oldJa || ja); });
  const tr = {};
  TERM_LANGS.forEach(function (l) {
    const given = clean_(q[l]);
    tr[l] = given || (cur && cur.ja === ja && cur[l]) || clean_(deepl_('post', '/v2/translate', { text: [ja], source_lang: 'JA', target_lang: LANG[l].tgt }).translations[0].text);
  });
  return withLock_(function () {
    const sh = termSheet_(), t = terms_();
    const hit = t.find(function (x) { return x.ja === clean_(q.oldJa || ja); });
    const row = [safe_(ja), safe_(tr.en), safe_(q.memo != null ? q.memo : (hit ? hit.memo : '')), safe_(tr.zh), safe_(tr.tw), safe_(tr.ko)];
    if (hit) sh.getRange(hit.row, 1, 1, 6).setValues([row]); else sh.appendRow(row);
    props_().setProperty('GLOSS_DIRTY', '1');
    return { ja: ja, en: tr.en, zh: tr.zh, tw: tr.tw, ko: tr.ko };
  });
}
function deleteTerm_(q) {
  return withLock_(function () {
    const hit = terms_().find(function (x) { return x.ja === clean_(q.ja); });
    if (hit) termSheet_().deleteRow(hit.row);
    props_().setProperty('GLOSS_DIRTY', '1');
    return {};
  });
}
// 用語集の番号（方向ごと。例 'ja-zh'）
function glossIds_() {
  const P = props_();
  let ids = {};
  try { ids = JSON.parse(P.getProperty('GLOSS_IDS') || '{}'); } catch (e) {}
  if (!ids['ja-en'] && P.getProperty('GLOSS_JA_EN')) ids['ja-en'] = P.getProperty('GLOSS_JA_EN');
  if (!ids['en-ja'] && P.getProperty('GLOSS_EN_JA')) ids['en-ja'] = P.getProperty('GLOSS_EN_JA');
  return ids;
}
function glossId_(dir) { return glossIds_()[dir] || null; }
// DeepLの用語集を作り直す。無料版は用語集が1つまでなので、
// 1つの用語集に言語の組み合わせごとの辞書をまとめる（DeepLの多言語用語集 v3）。
// 繁体字は簡体字と辞書を分けられないため、用語集を使わない。
function syncGlossary_() {
  const P = props_();
  const t = terms_();
  const del = {}, old = glossIds_();
  Object.keys(old).forEach(function (k) { del[old[k]] = 1; });
  try {
    (deepl_('get', '/v3/glossaries').glossaries || []).forEach(function (g) { if (/^宿/.test(String(g.name))) del[g.glossary_id] = 1; });
  } catch (e) {}
  Object.keys(del).forEach(function (id) {
    try { deepl_('delete', '/v3/glossaries/' + id); } catch (e) { try { deepl_('delete', '/v2/glossaries/' + id); } catch (e2) {} }
  });
  P.deleteProperty('GLOSS_JA_EN'); P.deleteProperty('GLOSS_EN_JA'); P.deleteProperty('GLOSS_IDS');
  let pairs = {};
  try {
    (deepl_('get', '/v2/glossary-language-pairs').supported_languages || []).forEach(function (p) {
      pairs[String(p.source_lang).toLowerCase() + '>' + String(p.target_lang).toLowerCase()] = 1;
    });
  } catch (e) { pairs = { 'ja>en': 1, 'en>ja': 1 }; }
  const tsv = function (list) {
    const seen = {};
    return list.map(function (p) { return [clean_(p[0]), clean_(p[1])]; })
      .filter(function (p) { if (!p[0] || !p[1] || seen[p[0]]) return false; seen[p[0]] = 1; return true; })
      .map(function (p) { return p[0] + '\t' + p[1]; }).join('\n');
  };
  const dicts = [], dirs = [], skipped = [], errs = [];
  ['en', 'zh', 'ko'].forEach(function (l) {
    const rows = t.filter(function (x) { return x[l]; });
    if (!rows.length) return;
    [['ja', l], [l, 'ja']].forEach(function (d) {
      if (!pairs[d[0] + '>' + d[1]]) { skipped.push(d[0] + '-' + d[1]); return; }
      const entries = tsv(rows.map(function (x) { return d[0] === 'ja' ? [x.ja, x[l]] : [x[l], x.ja]; }));
      if (!entries) return;
      dicts.push({ source_lang: d[0], target_lang: d[1], entries: entries, entries_format: 'tsv' });
      dirs.push(d[0] + '-' + d[1]);
    });
  });
  const ids = {};
  if (dicts.length) {
    try {
      const r = deepl_('post', '/v3/glossaries', { name: '宿の翻訳アプリ 用語集', dictionaries: dicts });
      dirs.forEach(function (d) { ids[d] = r.glossary_id; });
    } catch (e) { errs.push(String(e.message).slice(0, 200)); }
  }
  P.setProperty('GLOSS_IDS', JSON.stringify(ids));
  P.setProperty('GLOSS_SYNC', JSON.stringify({ at: Date.now(), n: t.length, dirs: Object.keys(ids), error: errs[0] || '' }));
  P.deleteProperty('GLOSS_DIRTY');
  return { n: t.length, dirs: Object.keys(ids), skipped: skipped, errors: errs };
}

function saveRule_(q) {
  const ja = String(q.ja || '').trim();
  if (!ja) throw new Error('日本語のルールを入れてください');
  const cur = q.id ? rules_().find(function (r) { return r.id === q.id; }) : null;
  const jaChanged = !cur || cur.ja !== ja || q.retranslate;
  // 訳文：手で直した訳があればそれを使い、日本語が変わった言語は訳し直す
  const tr = {};
  RULE_LANGS.forEach(function (l) {
    const given = q[l] != null ? String(q[l]).trim() : '';
    const changedByHand = cur && given && given !== cur[l];
    tr[l] = changedByHand ? given : (jaChanged || !given) ? translateRule_(ja, l) : given;
  });
  // 読み上げ用の短い文：日本語が変わった時だけ訳し直す
  const sayJa = q.sayJa != null ? String(q.sayJa).trim() : (cur && cur.say.ja) || '';
  let say = {};
  if (sayJa) {
    if (cur && cur.say.ja === sayJa && !q.retranslate) say = cur.say;
    else { say = { ja: sayJa }; RULE_LANGS.forEach(function (l) { say[l] = translateRule_(sayJa, l); }); }
  }
  const photos = Array.isArray(q.photos) ? q.photos.map(String).filter(function (x) { return /^[\w-]{10,}$/.test(x); }) : (cur ? cur.photos : []);
  const memo = q.memo != null ? String(q.memo) : (cur ? cur.memo : '');
  const show = q.show != null ? !!q.show : (cur ? cur.show : true);
  const from = q.from != null ? md_(q.from) : (cur ? cur.from : ''), to = q.to != null ? md_(q.to) : (cur ? cur.to : '');
  const res = withLock_(function () {
    const sh = ruleSheet_(), all = rules_();
    const hit = q.id ? all.find(function (r) { return r.id === q.id; }) : null;
    const order = hit ? hit.order : (all.reduce(function (m, r) { return Math.max(m, r.order); }, 0) + 1);
    const row = [hit ? hit.id : Utilities.getUuid().slice(0, 8), q.cat || (hit ? hit.cat : 'その他'), safe_(ja), safe_(tr.en), safe_(tr.zh), safe_(tr.tw), safe_(tr.ko), order, new Date(),
      photos.join(','), safe_(memo), show ? '1' : '0', from ? "'" + from : '', to ? "'" + to : '', sayJa ? JSON.stringify(say) : ''];
    if (hit) sh.getRange(hit.row, 1, 1, row.length).setValues([row]); else sh.appendRow(row);
    return { id: row[0], en: tr.en, zh: tr.zh, tw: tr.tw, ko: tr.ko };
  });
  // 外した写真はドライブのゴミ箱へ
  if (cur) trashPhotos_(cur.photos.filter(function (x) { return photos.indexOf(x) < 0; }));
  return res;
}

/* ---------- 写真の文字読み取り（Googleドライブの文字認識を利用） ----------
 * Apps Script の「サービス」で Drive API を追加しておくこと。
 * 写真を一時的にGoogleドキュメントへ変換して文字を取り出し、すぐ削除する。 */
function ocr_(q) {
  const b64 = String(q.image || '').replace(/^data:[^,]+,/, '');
  if (!b64) throw new Error('写真がありません');
  if (b64.length > 8000000) throw new Error('写真が大きすぎます');
  const blob = Utilities.newBlob(Utilities.base64Decode(b64), 'image/jpeg', 'ocr.jpg');
  const opt = {};
  if (/^(ja|en|zh|ko)$/.test(String(q.hint || ''))) opt.ocrLanguage = q.hint;
  const f = Drive.Files.create({ name: '翻訳アプリ読み取り_' + Date.now(), mimeType: 'application/vnd.google-apps.document' }, blob, opt);
  try {
    const text = DocumentApp.openById(f.id).getBody().getText().replace(/^\s+|\s+$/g, '');
    if (!text) throw new Error('文字が読み取れませんでした。明るい場所で、文字が大きく写るように撮ってください');
    return { text: text };
  } finally {
    try { Drive.Files.remove(f.id); } catch (e) { try { DriveApp.getFileById(f.id).setTrashed(true); } catch (e2) {} }
  }
}

function logTurn_(dir, src, tgt) {

    sh.appendRow([Utilities.getUuid().slice(0, 8), dir, safe_(src), safe_(tgt), String(Date.now())]);
    const n = sh.getLastRow() - 1;
    if (n > HIST_MAX) sh.deleteRows(2, n - HIST_MAX);
    if (!autoOn_()) return false;
    const nt = norm_(src);
    const same = hist_().filter(function (h) { return h.dir === dir && norm_(h.src) === nt; });
    if (same.length < AUTO_N) return false;
    // 一番多く出た訳を採用（同数なら最新）

    dismiss_({ ids: same.map(function (h) { return h.id; }) });
    return true;
  } catch (e) { return false; /* 履歴の保存に失敗しても翻訳は返す */ }

function logFolder_() {
  const P = props_();
  let id = P.getProperty('LOG_FOLDER');
  if (id) { try { const f = Drive.Files.get(id, { fields: 'id,trashed' }); if (!f.trashed) return id; } catch (e) {} }
  id = Drive.Files.create({ name: 'おもてなし通訳_会話ログ', mimeType: 'application/vnd.google-apps.folder' }).id;
  P.setProperty('LOG_FOLDER', id);

}
function csv_(rows) {
  return rows.map(function (r) {
    return r.map(function (c) { c = String(c == null ? '' : c).replace(/^'/, ''); return /[",\n\r]/.test(c) ? '"' + c.replace(/"/g, '""') + '"' : c; }).join(',');
  }).join('\r\n');
}
// 今月より前の行をCSVにしてドライブへ移し、シートからは消す
function archiveLog_() {
  return withLock_(function () {
    const sh = sheet_(SH_LOG, HEAD_LOG);
    const v = sh.getDataRange().getValues();
    if (v.length < 2) return { archived: 0, files: [] };
    const cur = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyy-MM');
    const byMonth = {}, keep = [];
    v.slice(1).forEach(function (r) {
      const m = r[1] instanceof Date ? Utilities.formatDate(r[1], 'Asia/Tokyo', 'yyyy-MM') : String(r[1]).replace(/^'/, '');
      if (m && m < cur) (byMonth[m] = byMonth[m] || []).push(r); else keep.push(r);
    });
    const months = Object.keys(byMonth).sort();
    if (!months.length) return { archived: 0, files: [] };
    const folder = logFolder_(), files = [];
    months.forEach(function (m) {
      let name = 'log_' + m + '.csv', n = 2;
      while (Drive.Files.list({ q: "name='" + name + "' and '" + folder + "' in parents and trashed=false", fields: 'files(id)' }).files.length) name = 'log_' + m + '_' + (n++) + '.csv';
      const blob = Utilities.newBlob('﻿' + csv_([HEAD_LOG].concat(byMonth[m])), 'text/csv', name);
      Drive.Files.create({ name: name, parents: [folder], mimeType: 'text/csv' }, blob);
      files.push(name);
    });
    // シートを今月分だけにする
    const n = v.length - 1;
    sh.getRange(2, 1, n, HEAD_LOG.length).clearContent();
    if (keep.length) sh.getRange(2, 1, keep.length, HEAD_LOG.length).setValues(keep);
    if (sh.getLastRow() > keep.length + 1) sh.deleteRows(keep.length + 2, sh.getLastRow() - keep.length - 1);
    return { archived: n - keep.length, files: files };
  });
}

// チェックアウト日の翌日正午まで使える
function stayValid_(st) {
  if (!st || st.state !== 'open') return false;
  if (!/^\d{4}-\d{2}-\d{2}$/.test(st.checkout)) return true;
  const until = new Date(st.checkout + 'T12:00:00+09:00').getTime() + 24 * 3600 * 1000;
  return Date.now() < until;
}
function createStay_(q) {
  const lang = GUEST_LANGS.indexOf(q.lang) >= 0 ? q.lang : 'en';

  const id = Utilities.getUuid().replace(/-/g, '').slice(0, 20);
  sheet_(SH_STAY, HEAD_STAY).appendRow([id, safe_(clean_(q.memo).slice(0, 60)), lang, "'" + co, Date.now(), 'open', 0, 0, 0]);
  return { id: id };
}

/* ---------- ゲスト側（合言葉で動く。パスコード不要） ---------- */
function guestApi_(q) {
  try {
    const st = stays_().find(function (x) { return x.id === String(q.token || ''); });
    if (!stayValid_(st)) return { ok: false, expired: true, error: 'This link has expired. / このリンクは使えません' };
    switch (q.action) {
      case 'gInfo': return Object.assign({ ok: true }, gInfo_(st, q));
      case 'gSend': return Object.assign({ ok: true }, gSend_(st, q));
      case 'gPoll': return Object.assign({ ok: true }, gPoll_(st, q));
      case 'gTranslate': return Object.assign({ ok: true }, gTranslate_(st, q));
      default: return { ok: false, error: 'unknown' };
    }
  } catch (e) { return { ok: false, error: String((e && e.message) || e) }; }
}
function gLang_(st, q) { return GUEST_LANGS.indexOf(q.lang) >= 0 ? q.lang : st.lang; }
function gInfo_(st, q) {
  const lang = gLang_(st, q), inf = info_();
  const rules = rules_().filter(guestVisible_).sort(function (a, b) { return a.order - b.order; })
    .map(function (r) { return { cat: r.cat, ja: r.ja, tr: r[lang] || r.en, say: r.say[lang] || '', photos: r.photos }; });
  return { lang: st.lang, checkout: st.checkout, store: { name: inf.name, addr: inf.addr, phone: inf.phone, map: inf.map, emergency: inf.emergency },
    hours: staffHours_(), rules: rules, messages: gMsgs_(st, 0),
    phrases: phrases_().map(function (p) { return { ja: p.ja, tr: p[lang] || p.en }; }).filter(function (p) { return p.tr; }) };
}
function gMsgs_(st, since) {
  return msgs_().filter(function (m) { return m.stay === st.id && m.t > since; })
    .map(function (m) { return m.from === 'guest' ? { t: m.t, from: 'guest', text: m.text, ja: m.tr } : { t: m.t, from: m.from, text: m.tr, ja: m.text }; });
}
function gPoll_(st, q) { return { messages: gMsgs_(st, Number(q.since) || 0), hours: staffHours_() }; }
// 1日の使用文字数（ゲストごと）
function gQuota_(st, n) {
  const P = props_(), key = 'GQ_' + st.id, day = Utilities.formatDate(new Date(), 'Asia/Tokyo', 'yyyyMMdd');
  let o = {};
  try { o = JSON.parse(P.getProperty(key) || '{}'); } catch (e) {}
  if (o.d !== day) o = { d: day, n: 0 };
  if (o.n + n > GUEST_DAILY_CHARS) throw new Error('Daily translation limit reached. / 本日の翻訳の上限に達しました');
  o.n += n; P.setProperty(key, JSON.stringify(o));
}
function gSend_(st, q) {
  const lang = gLang_(st, q);
  const text = String(q.text || '').trim().slice(0, 1000);
  if (!text) throw new Error('empty');
  const ja = gTr_(st, lang, true, text, 'ゲストチャット');
  const hours = staffHours_();
  withLock_(function () {
    addMsg_(st.id, 'guest', text, lang, ja);
    // 通知はまとめる：同じゲストから15分以内に続いたら通知しない
    const now = Date.now(), notify = now - st.notified > 15 * 60 * 1000;
    bumpStay_(st, 1, notify ? now : st.notified);
    if (notify) linePush_('【おもてなし通訳】' + (st.memo || 'ゲスト') + 'さん（' + langName_(lang) + '）から\n「' + ja.slice(0, 200) + '」' +
      (hours.now ? '' : '\n※対応時間外のため、ゲストには自動で案内しています') + '\nアプリの「ゲスト」タブで返信してください');
    // 対応時間外は自動で案内（同じゲストに1時間に1回まで）
    if (!hours.now) {
      const last = msgs_().filter(function (m) { return m.stay === st.id && m.from === 'auto'; }).pop();
      if (!last || now - last.t > 3600 * 1000) {
        const inf = info_();
        const jaMsg = 'スタッフの対応時間は' + hours.open + '〜' + hours.close + 'です。対応時間になりしだい返信します。' +
          (inf.emergency ? '緊急の場合は ' + inf.emergency + ' へお電話ください。' : '') + '急病・事故・火事のときは、画面上の「SOS」から110番・119番に電話できます。このページの翻訳は今もお使いいただけます。';
        addMsg_(st.id, 'auto', jaMsg, 'ja', autoMsg_(jaMsg, lang));
      }
    }
  });
  return { messages: gMsgs_(st, Number(q.since) || 0), hours: hours };
}
// 時間外の案内文は言語ごとに一度だけ訳して使い回す
function autoMsg_(ja, lang) {
  const P = props_(), key = 'AUTO_MSG_' + lang;
  let o = {};
  try { o = JSON.parse(P.getProperty(key) || '{}'); } catch (e) {}
  if (o.ja === ja) return o.tr;
  const tr = deeplT_({ text: [ja], source_lang: 'JA', target_lang: LANG[lang].tgt }).translations[0].text;
  P.setProperty(key, JSON.stringify({ ja: ja, tr: tr }));
  return tr;
}
// ゲストが自分で使う翻訳（現地の人に見せる・現地の人の日本語を読む）
function gTranslate_(st, q) {
  const lang = gLang_(st, q);
  const text = String(q.text || '').trim().slice(0, 1000);
  if (!text) throw new Error('empty');
  return { text: gTr_(st, lang, q.dir !== 'from-ja', text, 'ゲストの翻訳') };
}
// ゲスト側の翻訳はスタッフ側と同じく、宿が覚えた用例 → 用語集つきのDeepL の順に使う。
// 同じ文を10分以内にもう一度訳すとき（訳してからスタッフに送るなど）は、前の訳を使い回す。
function gTr_(st, lang, toJa, text, where) {
  const dir = toJa ? lang + '-ja' : 'ja-' + lang;
  const nt = norm_(text);
  const hit = examples_().find(function (r) { return r.dir === dir && norm_(r.src) === nt; });
  if (hit) { bump_(hit); stat_('mem'); alog_(dir, text, hit.tgt, 'memory', where); return hit.tgt; }
  const cache = CacheService.getScriptCache();
  const key = 'GT_' + Utilities.base64EncodeWebSafe(Utilities.computeDigest(Utilities.DigestAlgorithm.MD5, st.id + '|' + dir + '|' + nt, Utilities.Charset.UTF_8));
  const old = cache.get(key);
  if (old) { alog_(dir, text, old, 'deepl', where); return old; }
  gQuota_(st, text.length);
  const body = toJa ? { text: [text], source_lang: LANG[lang].src, target_lang: 'JA' } : { text: [text], source_lang: 'JA', target_lang: LANG[lang].tgt };
  const gid = glossId_(dir); if (gid) body.glossary_id = gid;
  const out = deeplT_(body).translations[0].text;
  stat_('deepl');
  cache.put(key, out, 600);
  alog_(dir, text, out, 'deepl', where);
  return out;
}

function savePhrase_(q) {

/* ---------- LINE公式アカウントへの通知（Messaging API） ----------
 * スクリプトプロパティ LINE_TOKEN にチャネルアクセストークン（長期）を入れる（管理者が手で設定）。
 * 通知先は、公式アカウントをスタッフのLINEグループに招待するか、公式アカウントに「通知登録」と送ると候補に入り、
 * アプリの「ゲスト」タブで確定する。 */
function lineHook_(body) {
  const P = props_();
  let pend = [];
  try { pend = JSON.parse(P.getProperty('LINE_PENDING') || '[]'); } catch (e) {}
  (body.events || []).forEach(function (ev) {
    const src = ev.source || {};
    const id = src.groupId || src.roomId || src.userId;
    const type = src.groupId ? 'グループ' : src.roomId ? 'トークルーム' : '個人';
    const isJoin = ev.type === 'join';
    const isReg = ev.type === 'message' && ev.message && ev.message.type === 'text' && /通知登録/.test(ev.message.text);
    if (!id || !(isJoin || isReg)) return;
    pend = pend.filter(function (x) { return x.id !== id; });
    pend.push({ id: id, type: type, at: Date.now() });
    if (ev.replyToken && P.getProperty('LINE_TOKEN')) {
      lineCall_('/v2/bot/message/reply', { replyToken: ev.replyToken, messages: [{ type: 'text', text: 'おもてなし通訳：通知先の候補に登録しました。アプリの「ゲスト」タブ→「LINE通知」で確定してください。' }] });
    }
  });
  P.setProperty('LINE_PENDING', JSON.stringify(pend.slice(-5)));
}
function lineCall_(path, payload) {
  const token = String(props_().getProperty('LINE_TOKEN') || '').trim();
  if (!token) return { skipped: true };
  const res = UrlFetchApp.fetch('https://api.line.me' + path, { method: 'post', contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token }, payload: JSON.stringify(payload), muteHttpExceptions: true });
  return { code: res.getResponseCode(), body: res.getContentText().slice(0, 200) };
}
function linePush_(text) {
  const to = props_().getProperty('LINE_TO');
  if (!to) return { skipped: true };
  try { return lineCall_('/v2/bot/message/push', { to: to, messages: [{ type: 'text', text: text.slice(0, 4900) }] }); } catch (e) { return { error: String(e) }; }
}
function lineStatus_() {
  const P = props_();
  let pend = [];
  try { pend = JSON.parse(P.getProperty('LINE_PENDING') || '[]'); } catch (e) {}
  const to = P.getProperty('LINE_TO') || '';
  return { token: !!P.getProperty('LINE_TOKEN'), target: to ? { type: P.getProperty('LINE_TO_TYPE') || '', id: to.slice(0, 6) + '…' } : null,
    pending: pend.map(function (x) { return { key: x.id.slice(-6), type: x.type, at: x.at }; }) };
}
function lineSetTarget_(q) {
  const P = props_();
  let pend = [];
  try { pend = JSON.parse(P.getProperty('LINE_PENDING') || '[]'); } catch (e) {}
  if (q.clear) { P.deleteProperty('LINE_TO'); P.deleteProperty('LINE_TO_TYPE'); return lineStatus_(); }
  const hit = pend.find(function (x) { return x.id.slice(-6) === String(q.key); });
  if (!hit) throw new Error('候補が見つかりません');
  P.setProperty('LINE_TO', hit.id); P.setProperty('LINE_TO_TYPE', hit.type);
  P.setProperty('LINE_PENDING', JSON.stringify(pend.filter(function (x) { return x.id !== hit.id; })));
  return lineStatus_();
}
function lineTest_() {
  const r = linePush_('おもてなし通訳：通知のテストです。ゲストからメッセージが届くと、ここに通知します。');
  if (r.skipped) throw new Error('通知先またはトークンが未設定です');
  if (r.code && r.code !== 200) throw new Error('LINEの送信に失敗しました（' + r.code + '）' + (r.code === 429 ? '：今月の無料通数を超えた可能性があります' : ''));
  return { sent: true };
}
